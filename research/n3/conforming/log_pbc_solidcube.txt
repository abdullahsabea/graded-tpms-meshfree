# coding=utf-8
# Periodic-BC apparent C1111 on a body-fitted mesh (energy basis).
# u = eps_bar x + w with w periodic: the macro strain enters as a uniform initial
# strain (PRE_EPSI), so the unknown is w alone and opposite faces are tied by
# w(slave) = w(master image) with LIAISON_MAIL (non-matching triangulations).
# Slave sets exclude nodes already slaved by an earlier pair (edges/corners).
# The energy sign of eps_bar is irrelevant (U is even in eps_bar).
from code_aster.Commands import *
import numpy as np

DEBUT()

EPS = 1.0e-3
mesh0 = LIRE_MAILLAGE(FORMAT='ASTER', UNITE=20)
mesh = CREA_MAILLAGE(MAILLAGE=mesh0, LINE_QUAD=_F(TOUT='OUI'))
FACES = ('X0', 'X1', 'Y0', 'Y1', 'Z0', 'Z1')
mesh = DEFI_GROUP(reuse=mesh, MAILLAGE=mesh, CREA_GROUP_NO=tuple(_F(GROUP_MA=g) for g in FACES))
mesh = DEFI_GROUP(reuse=mesh, MAILLAGE=mesh, CREA_GROUP_NO=(
    _F(NOM='SY1', DIFFE=('Y1', 'X1')),
    _F(NOM='SZ1A', DIFFE=('Z1', 'X1')),
))
mesh = DEFI_GROUP(reuse=mesh, MAILLAGE=mesh, CREA_GROUP_NO=(
    _F(NOM='SZ1', DIFFE=('SZ1A', 'Y1')),
    _F(NOM='PIN', GROUP_NO='X0', NUME_INIT=1, NUME_FIN=1),  # one node: removes rigid translation
))

model = AFFE_MODELE(MAILLAGE=mesh, AFFE=_F(MODELISATION='3D', PHENOMENE='MECANIQUE', GROUP_MA='SOLID'))
mat = DEFI_MATERIAU(ELAS=_F(E=1.0, NU=0.3, RHO=1.0))
LAM, MU = 0.3 / (1.3 * 0.4), 1.0 / 2.6
C11 = LAM + 2.0 * MU
chmat = AFFE_MATERIAU(MAILLAGE=mesh, AFFE=_F(GROUP_MA='SOLID', MATER=mat))

per = AFFE_CHAR_MECA(MODELE=model, LIAISON_MAIL=(
    _F(GROUP_MA_MAIT='SOLID', GROUP_NO_ESCL='X1', TRAN=(-1.0, 0.0, 0.0)),
    _F(GROUP_MA_MAIT='SOLID', GROUP_NO_ESCL='SY1', TRAN=(0.0, -1.0, 0.0)),
    _F(GROUP_MA_MAIT='SOLID', GROUP_NO_ESCL='SZ1', TRAN=(0.0, 0.0, -1.0)),
))
pin = AFFE_CHAR_MECA(MODELE=model, DDL_IMPO=_F(GROUP_NO='PIN', DX=0.0, DY=0.0, DZ=0.0))
load = AFFE_CHAR_MECA(MODELE=model, PRE_EPSI=_F(GROUP_MA='SOLID', EPXX=EPS))

res = MECA_STATIQUE(MODELE=model, CHAM_MATER=chmat,
                    EXCIT=(_F(CHARGE=per), _F(CHARGE=pin), _F(CHARGE=load)),
                    SOLVEUR=_F(METHODE='MUMPS', STOP_SINGULIER='NON', RESI_RELA=1.0e-6))  # sliver tets on cut faces trip the 8-digit pivot test; Hill-Mandel route agreement checks the solution
# PRE_EPSI enters as an equivalent load; code_aster reports sigma(u) = C eps(u)
# and ENER_ELAS = 1/2 int eps(u):C:eps(u). The true field is C(eps(u) - eps0):
#   stress route : C1111 = |int sigma11(u) - C11 eps0 Vs| / eps0
#   energy route : U = E1 - eps0 int sigma11(u) + 1/2 C11 eps0^2 Vs,  C1111 = 2U/eps0^2
# Hill-Mandel requires the two to agree under periodic BC.
res = CALC_CHAMP(reuse=res, RESULTAT=res, CONTRAINTE=('SIEF_ELGA',))
tab = POST_ELEM(RESULTAT=res, MODELE=model,
                INTEGRALE=_F(GROUP_MA='SOLID', NOM_CHAM='SIEF_ELGA', NOM_CMP='SIXX', TYPE_MAILLE='3D'))
vals = tab.EXTR_TABLE().values()
S = float(np.asarray(vals[[k for k in vals if k.startswith('INTE_SIXX')][0]], dtype=float)[-1])
ener = POST_ELEM(RESULTAT=res, MODELE=model, ENER_ELAS=_F(GROUP_MA='SOLID'))
E1 = float(np.asarray(ener.EXTR_TABLE().values()['TOTALE'], dtype=float)[-1])
mass = POST_ELEM(MODELE=model, CHAM_MATER=chmat, MASS_INER=_F(GROUP_MA='SOLID'))
Vs = float(np.asarray(mass.EXTR_TABLE().values()['MASSE'], dtype=float)[-1])
c_stress = abs(S - C11 * EPS * Vs) / EPS
U = E1 - EPS * S + 0.5 * C11 * EPS**2 * Vs
print('SOLID_VOLUME = %.10e' % Vs)
print('C1111_PBC    = %.10e' % c_stress)
print('C1111_PBC_EN = %.10e' % (2.0 * U / EPS**2))

FIN()
