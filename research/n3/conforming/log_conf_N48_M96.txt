# coding=utf-8
# KUBC energy-basis homogenization on a body-fitted tetrahedral mesh.
# Physics identical to homog2.comm (u = eps*x on the whole cell boundary,
# C1111 = 2U/(V eps^2)); linear TETRA4 from TetGen are upgraded to quadratic
# TETRA10 here, and the boundary node group is built from the cap faces.
from code_aster.Commands import *
import numpy as np

DEBUT()

EPS = 1.0e-3

mesh0 = LIRE_MAILLAGE(FORMAT='ASTER', UNITE=20)
mesh = CREA_MAILLAGE(MAILLAGE=mesh0, LINE_QUAD=_F(TOUT='OUI'))
FACES = ('X0', 'X1', 'Y0', 'Y1', 'Z0', 'Z1')
mesh = DEFI_GROUP(reuse=mesh, MAILLAGE=mesh,
                  CREA_GROUP_NO=tuple(_F(GROUP_MA=g) for g in FACES))

model = AFFE_MODELE(MAILLAGE=mesh,
                    AFFE=_F(MODELISATION='3D', PHENOMENE='MECANIQUE', GROUP_MA='SOLID'))

mat = DEFI_MATERIAU(ELAS=_F(E=1.0, NU=0.3, RHO=1.0))
chmat = AFFE_MATERIAU(MAILLAGE=mesh, AFFE=_F(GROUP_MA='SOLID', MATER=mat))

ux = FORMULE(VALE='0.001*X', NOM_PARA=('X', 'Y', 'Z'))
f0 = FORMULE(VALE='0.0', NOM_PARA=('X', 'Y', 'Z'))
bc = AFFE_CHAR_MECA_F(MODELE=model,
                      DDL_IMPO=_F(GROUP_NO=FACES, DX=ux, DY=f0, DZ=f0))

res = MECA_STATIQUE(MODELE=model, CHAM_MATER=chmat, EXCIT=_F(CHARGE=bc),
                    SOLVEUR=_F(METHODE='MUMPS'))

ener = POST_ELEM(RESULTAT=res, MODELE=model, ENER_ELAS=_F(GROUP_MA='SOLID'))
U = float(np.asarray(ener.EXTR_TABLE().values()['TOTALE'], dtype=float)[-1])
mass = POST_ELEM(MODELE=model, CHAM_MATER=chmat, MASS_INER=_F(GROUP_MA='SOLID'))
V = float(np.asarray(mass.EXTR_TABLE().values()['MASSE'], dtype=float)[-1])
print('U_ELASTIC    = %.10e' % U)
print('SOLID_VOLUME = %.10e' % V)
print('C1111_ENERGY = %.10e' % (2.0 * U / EPS**2))

FIN()
